cmake_minimum_required(VERSION 3.28)
project(DemoTracerKHookTests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(MMS "$ENV{MMSOURCE_DEV}")
include(../khook.cmake)
set(KHOOK_ROOT "${MMS}/third_party/khook")

# Exercise the real upstream engine in tests only. Product DLLs use Metamod.
set(SAFETYHOOK_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(SAFETYHOOK_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${KHOOK_ROOT}/third_party/safetyhook" safetyhook)
add_library(DemoTracerKHookTestEngine SHARED
    "${KHOOK_ROOT}/src/detour.cpp" "${KHOOK_ROOT}/src/ranges.cpp")
target_include_directories(DemoTracerKHookTestEngine PUBLIC "${KHOOK_ROOT}/include")
target_compile_definitions(DemoTracerKHookTestEngine PRIVATE KHOOK_STANDALONE KHOOK_EXPORT)
target_link_libraries(DemoTracerKHookTestEngine PRIVATE safetyhook)
if(WIN32)
    target_compile_definitions(DemoTracerKHookTestEngine PRIVATE NOMINMAX)
else()
    set_target_properties(safetyhook Zydis Zycore PROPERTIES POSITION_INDEPENDENT_CODE ON)
endif()
add_executable(DemoTracerKHookTests khook_tests.cpp)
target_compile_definitions(DemoTracerKHookTests PRIVATE KHOOK_STANDALONE)
target_link_libraries(DemoTracerKHookTests PRIVATE DemoTracerKHookTestEngine)
enable_testing()
add_test(NAME DemoTracerKHookTests COMMAND DemoTracerKHookTests)
set_tests_properties(DemoTracerKHookTests PROPERTIES TIMEOUT 30)

add_executable(DemoTracerSignatureTests sig_scan_tests.cpp ../sig_scan.cpp)
target_include_directories(DemoTracerSignatureTests PRIVATE ../../third_party)
target_compile_definitions(DemoTracerSignatureTests PRIVATE KHOOK_STANDALONE)
target_link_libraries(DemoTracerSignatureTests PRIVATE DemoTracerKHookTestEngine)
if(WIN32)
    target_compile_definitions(DemoTracerSignatureTests PRIVATE NOMINMAX)
    target_link_libraries(DemoTracerSignatureTests PRIVATE psapi)
else()
    target_link_libraries(DemoTracerSignatureTests PRIVATE dl)
endif()
add_test(NAME DemoTracerSignatureTests COMMAND DemoTracerSignatureTests)
